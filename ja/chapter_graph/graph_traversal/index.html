9.3   グラフの探索¶

木が表すのは「一対多」の関係ですが、グラフにはさらに自由度があり、任意の「多対多」の関係を表せます。このため木はグラフの特殊な場合と見なせます。同じように、木の探索もグラフの探索の特殊な場合です。

グラフでも木でも、すべての頂点を訪問するには探索アルゴリズムを使います。グラフの探索も、幅優先探索と深さ優先探索の 2 種類に分けられます。

9.3.1   幅優先探索¶

幅優先探索では、ある頂点から出発し、常に始点に近い頂点を優先して訪問しながら、外側へ 1 層ずつ探索範囲を広げます。図 9-9 では左上の頂点から出発し、まずその隣接頂点をすべて調べ、次の頂点についても隣接頂点をすべて調べます。これを全頂点の訪問が終わるまで繰り返します。

1.   アルゴリズムの実装¶

BFS は通常、次のコードのようにキューを使って実装します。先に入ったものが先に出るキューの「先入れ先出し」という性質は、BFS で近い頂点から遠い頂点へと訪問する考え方に対応します。

  1. 探索の始点 startVet をキューに入れ、ループを開始します。
  2. 各反復ではキューの先頭から頂点を取り出して訪問を記録し、その頂点の隣接頂点をすべてキューの末尾に追加します。
  3. 全頂点の訪問が終わるまで、手順 2. を繰り返します。

同じ頂点を繰り返し訪問しないように、ハッシュ集合 visited を使って訪問済みの頂点を記録します。

ヒント

ハッシュ集合は、key だけを格納して value をもたないハッシュテーブルと考えられます。key の追加、削除、検索、更新を \(O(1)\) の時間計算量で行えます。key が重複しない性質を生かし、データの重複を取り除くときなどに使います。

graph_bfs.py
def graph_bfs(graph: GraphAdjList, start_vet: Vertex) -> list[Vertex]:
    """幅優先探索"""
    # 指定した頂点の隣接頂点をすべて取得できるよう、グラフを隣接リストで表す
    # 頂点の訪問順
    res = []
    # 訪問済みの頂点を記録するハッシュ集合
    visited = set[Vertex]([start_vet])
    # BFS にキューを使う
    que = deque[Vertex]([start_vet])
    # 頂点 vet から開始し、すべての頂点を訪問するまで繰り返す
    while len(que) > 0:
        vet = que.popleft()  # キューの先頭から頂点を取り出す
        res.append(vet)  # 訪問した頂点を記録する
        # この頂点の隣接頂点をすべて調べる
        for adj_vet in graph.adj_list[vet]:
            if adj_vet in visited:
                continue  # 訪問済みの頂点を飛ばす
            que.append(adj_vet)  # 未訪問の頂点だけをキューに追加する
            visited.add(adj_vet)  # この頂点を訪問済みにする
    # 頂点を訪問順に返す
    return res

// 指定した頂点の隣接頂点をすべて取得できるよう、グラフを隣接リストで表す
vector<Vertex *> graphBFS(GraphAdjList &graph, Vertex *startVet) {
    // 頂点の訪問順
    vector<Vertex *> res;
    // 訪問済みの頂点を記録するハッシュ集合
    unordered_set<Vertex *> visited = {startVet};
    // BFS にキューを使う
    queue<Vertex *> que;
    que.push(startVet);
    // 頂点 vet から開始し、すべての頂点を訪問するまで繰り返す
    while (!que.empty()) {
        Vertex *vet = que.front();
        que.pop();          // キューの先頭から頂点を取り出す
        res.push_back(vet); // 訪問した頂点を記録する
        // この頂点の隣接頂点をすべて調べる
        for (auto adjVet : graph.adjList[vet]) {
            if (visited.count(adjVet))
                continue;            // 訪問済みの頂点を飛ばす

            visited.emplace(adjVet); // この頂点を訪問済みにする
        }
    }
    // 頂点を訪問順に返す
    return res;
}

        }
    }

    // 頂点を訪問順に返す
    return res;
}

/* 幅優先探索 */
// 指定した頂点の隣接頂点をすべて取得できるよう、グラフを隣接リストで表す
func graphBFS(g *graphAdjList, startVet Vertex) []Vertex {
    // 頂点の訪問順
    res := make([]Vertex, 0)
    // 訪問済みの頂点を記録するハッシュ集合
    visited := make(map[Vertex]struct{})
    visited[startVet] = struct{}{}
    // BFS に使うキューをスライスで表す
    queue := make([]Vertex, 0)
    queue = append(queue, startVet)
    // 頂点 vet から開始し、すべての頂点を訪問するまで繰り返す
    for len(queue) > 0 {
        // キューの先頭から頂点を取り出す
        vet := queue[0]
        queue = queue[1:]
        // 訪問した頂点を記録する
        res = append(res, vet)
        // この頂点の隣接頂点をすべて調べる
        for _, adjVet := range g.adjList[vet] {
            _, isExist := visited[adjVet]
            // 未訪問の頂点だけをキューに追加する
            if !isExist {
                queue = append(queue, adjVet)
                visited[adjVet] = struct{}{}
            }
        }
    }
    // 頂点を訪問順に返す
    return res
}

fn graph_bfs(graph: GraphAdjList, start_vet: Vertex) -> Vec<Vertex> {
    // 頂点の訪問順
    let mut res = vec![];
    // 訪問済みの頂点を記録するハッシュ集合
    let mut visited = HashSet::new();

    // BFS にキューを使う

        res.push(vet); // 訪問した頂点を記録する

        // この頂点の隣接頂点をすべて調べる
        if let Some(adj_vets) = graph.adj_list.get(&vet) {
            for &adj_vet in adj_vets {
                if visited.contains(&adj_vet) {
                    continue; // 訪問済みの頂点を飛ばす
                }
                que.push_back(adj_vet); // 未訪問の頂点だけをキューに追加する
                visited.insert(adj_vet); // この頂点を訪問済みにする
            }
        }
    }
    // 頂点を訪問順に返す
    res
}

typedef struct {
    Vertex *vertices[MAX_SIZE];
    int front, rear, size;
} Queue;

/* コンストラクタ */
Queue *newQueue() {
    Queue *q = (Queue *)malloc(sizeof(Queue));
    q->front = q->rear = q->size = 0;
    return q;
}

/* キューが空か判定する */
int isEmpty(Queue *q) {
    return q->size == 0;
}

/* キューに追加する */
void enqueue(Queue *q, Vertex *vet) {
    q->vertices[q->rear] = vet;
    q->rear = (q->rear + 1) % MAX_SIZE;
    q->size++;
}

/* キューから取り出す */
Vertex *dequeue(Queue *q) {
    Vertex *vet = q->vertices[q->front];
    q->front = (q->front + 1) % MAX_SIZE;
    q->size--;
    return vet;
}

/* 頂点が訪問済みか調べる */
int isVisited(Vertex **visited, int size, Vertex *vet) {
    // ノードを順に探すため、O(n) 時間かかる
    for (int i = 0; i < size; i++) {
        if (visited[i] == vet)
            return 1;
    }
    return 0;
}

/* 幅優先探索 */
// 指定した頂点の隣接頂点をすべて取得できるよう、グラフを隣接リストで表す
void graphBFS(GraphAdjList *graph, Vertex *startVet, Vertex **res, int *resSize, Vertex **visited, int *visitedSize) {
    // BFS にキューを使う
    Queue *queue = newQueue();
    enqueue(queue, startVet);
    visited[(*visitedSize)++] = startVet;
    // 頂点 vet から開始し、すべての頂点を訪問するまで繰り返す
    while (!isEmpty(queue)) {
        Vertex *vet = dequeue(queue); // キューの先頭から頂点を取り出す
        res[(*resSize)++] = vet;      // 訪問した頂点を記録する
        // この頂点の隣接頂点をすべて調べる
        AdjListNode *node = findNode(graph, vet);
        while (node != NULL) {
            // 訪問済みの頂点を飛ばす
            if (!isVisited(visited, *visitedSize, node->vertex)) {
                enqueue(queue, node->vertex);             // 未訪問の頂点だけをキューに追加する
                visited[(*visitedSize)++] = node->vertex; // この頂点を訪問済みにする
            }
            node = node->next;
        }
    }
    // メモリを解放する
    free(queue);
}

幅優先探索の訪問順は一意ですか。

一意ではありません。幅優先探索で必要なのは、始点から近い順に訪問することだけで、同じ距離にある頂点同士の訪問順は任意に入れ替えられます。図 9-10 では、頂点 \(1\) と \(3\) の訪問順を交換でき、頂点 \(2\)、\(4\)、\(6\) の訪問順も自由に入れ替えられます。

2.   計算量の分析¶

時間計算量：各頂点をキューに 1 回ずつ入れて取り出すのに \(O(|V|)\) 時間かかります。隣接頂点を調べるとき、無向グラフではすべての辺を \(2\) 回ずつ調べるので、\(O(2|E|)\) 時間かかります。全体の所要時間は \(O(|V| + |E|)\) です。

空間計算量：リスト res、ハッシュ集合 visited、キュー que に入る頂点数はいずれも最大 \(|V|\) なので、使用する空間は \(O(|V|)\) です。

9.3.2   深さ優先探索¶

深さ優先探索では、まず進めるところまで進み、行き止まりに来たら引き返します。図 9-11 では左上の頂点から出発し、現在の頂点の隣接頂点を 1 つ選んで進みます。行き止まりで戻り、また進めるところまで進んで戻る操作を、すべての頂点を訪問するまで繰り返します。

    // この頂点の隣接頂点をすべて調べる
    for _, adjVet := range g.adjList[vet] {
        _, isExist := visited[adjVet]
        // 隣接頂点を再帰的に訪問する
        if !isExist {
            dfs(g, visited, res, adjVet)
        }
    }
}

void dfs(GraphAdjList *graph, Vertex **res, int *resSize, Vertex *vet) {
    // 訪問した頂点を記録する
    res[(*resSize)++] = vet;
    // この頂点の隣接頂点をすべて調べる
    AdjListNode *node = findNode(graph, vet);
    while (node != NULL) {
        // 訪問済みの頂点を飛ばす
        if (!isVisited(res, *resSize, node->vertex)) {
            // 隣接頂点を再帰的に訪問する
            dfs(graph, res, resSize, node->vertex);
        }
        node = node->next;
    }
}

  1. 始点 0 の訪問

    各ステップの図では、初めて訪問した順に頂点を res に記録し、visited で重複を防ぎます。波括弧は集合の要素を列挙するだけで、順序を表しません。最初の呼び出しで頂点 0 に入り（イベント 1）、まず res と visited に記録してから、隣接頂点を 1、3 の順に調べます。現在の呼び出し経路は 0 で、res = [0]、visited = {0} です。0 の隣接頂点 1 と 3 はどちらも未訪問です。

  2. 0 から 1 へ進む

    頂点 0 の未訪問の隣接頂点は 1 と 3 です。この例では先に再帰呼び出しで 1 に入り（イベント 2）、初めて訪問した 1 を res と visited に記録します。もう一方の 3 は、1 の枝から戻った後で探索します。現在の呼び出し経路は 0→1 で、res = [0, 1]、visited = {0, 1} です。

  3. 1 から 2 へ進む

    頂点 1 の隣接頂点 0 は visited にあるので飛ばします。再帰呼び出しで未訪問の隣接頂点 2 に入り（イベント 3）、2 を res と visited に記録します。現在の呼び出し経路は 0→1→2 で、res = [0, 1, 2]、visited = {0, 1, 2} です。

  4. 2 から 5 へ進む

    頂点 2 の隣接頂点 1 は訪問済みです。再帰呼び出しで未訪問の隣接頂点 5 に入り（イベント 4）、5 を res と visited に記録します。現在の呼び出し経路は 0→1→2→5 で、res = [0, 1, 2, 5]、visited = {0, 1, 2, 5} です。5 の隣接頂点 4 と 6 はまだ未訪問です。

  5. 5 から 4 へ進む

    頂点 5 の隣接頂点 2 は訪問済みです。残る未訪問の隣接頂点 4、6 のうち、この例では先に再帰呼び出しで 4 に入り（イベント 5）、4 を res と visited に記録します。現在の呼び出し経路は 0→1→2→5→4 で、res = [0, 1, 2, 5, 4]、visited = {0, 1, 2, 4, 5} です。頂点 3 と 6 はまだ未訪問です。

  6. 4 から 5 に戻る

    頂点 4 の唯一の隣接頂点 5 は訪問済みなので、4 の呼び出しを終えて 5 に戻ります（イベント 6）。訪問済みの 5 に戻っても、5 を再度 res に追加することはありません。現在の呼び出し経路は 0→1→2→5 で、res は [0, 1, 2, 5, 4]、visited は {0, 1, 2, 4, 5} のままです。5 の隣接頂点 6 と、0 の隣接頂点 3 はまだ未訪問です。

  7. 5 から 6 へ進む

    頂点 5 に戻り、残る未訪問の隣接頂点 6 を調べます。再帰呼び出しで 6 に入り（イベント 7）、6 を res と visited に記録します。現在の呼び出し経路は 0→1→2→5→6 で、res = [0, 1, 2, 5, 4, 6]、visited = {0, 1, 2, 4, 5, 6} です。グラフ全体で未訪問の頂点は 3 だけです。

  8. 6 から 5 に戻る

    頂点 6 の唯一の隣接頂点 5 は訪問済みなので、6 の呼び出しを終えて 5 に戻ります（イベント 8）。新たな訪問はありません。現在の呼び出し経路は 0→1→2→5 で、res = [0, 1, 2, 5, 4, 6]、visited = {0, 1, 2, 4, 5, 6} です。5 の隣接頂点 2、4、6 はすべて処理済みです。

  9. 呼び出しを順に戻って始点 0 へ

    5、2、1 の隣接頂点はすべて調べ終えたので、5→2→1→0 と順に戻ります。イベント 9 で 5 から 2 に、イベント 10 で 2 から 1 に、イベント 11 で 1 から 0 に戻ります。現在の呼び出し経路は 0 で、res = [0, 1, 2, 5, 4, 6]、visited = {0, 1, 2, 4, 5, 6} です。始点 0 には未訪問の隣接頂点 3 の枝が残っています。

  10. 0 から 3 へ進む

    0 の隣接頂点 1 の枝からすべて戻ったので、次に未訪問の隣接頂点 3 に入り（イベント 12）、3 を res と visited に記録します。現在の呼び出し経路は 0→3 で、res = [0, 1, 2, 5, 4, 6, 3]、visited = {0, 1, 2, 3, 4, 5, 6} です。これで 7 頂点すべてが訪問済みとなり、最後の呼び出しから戻る処理が残ります。

  11. すべての再帰呼び出しの終了

    頂点 3 の唯一の隣接頂点 0 は訪問済みなので、まず 3 から 0 に戻ります（イベント 13）。この時点で始点 0 の隣接頂点もすべて処理済みです。最初に行ったいちばん外側の呼び出しから戻り（イベント 14）、深さ優先探索が終了します。呼び出しスタックは空になり、最終状態は res = [0, 1, 2, 5, 4, 6, 3]、visited = {0, 1, 2, 3, 4, 5, 6} です。図の始点 0 には戻りを示すオレンジ色の印があります。

res に記録するのは各頂点を初めて訪問したときだけで、戻るときは追加しません。隣接頂点を調べる順序が変わると、深さ優先探索の訪問順も変わります。全頂点の訪問後も、残っている再帰呼び出しを終了させる必要があります。

深さ優先探索の訪問順は一意ですか。

幅優先探索と同様に、深さ優先探索の訪問順も一意ではありません。ある頂点から次にどの方向を探索してもよく、隣接頂点を調べる順序を入れ替えても深さ優先探索です。

木の探索を例にすると、「根 \(\rightarrow\) 左 \(\rightarrow\) 右」「左 \(\rightarrow\) 根 \(\rightarrow\) 右」「左 \(\rightarrow\) 右 \(\rightarrow\) 根」は、それぞれ行きがけ順、通りがけ順、帰りがけ順に対応します。どこを先に訪問するかは違っても、3 つとも深さ優先探索です。

2.   計算量の分析¶

時間計算量：各頂点を \(1\) 回ずつ訪問するのに \(O(|V|)\) 時間、各辺を \(2\) 回ずつ調べるのに \(O(2|E|)\) 時間かかります。全体の所要時間は \(O(|V| + |E|)\) です。

空間計算量：リスト res とハッシュ集合 visited の頂点数はいずれも最大 \(|V|\) で、再帰の深さは最大 \(|V|\) なので、使用する空間は \(O(|V|)\) です。
