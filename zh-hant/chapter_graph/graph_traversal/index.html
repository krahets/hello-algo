9.3   圖的走訪¶

樹代表「一對多」的關係，圖的連接方式更自由，可以表示任意的「多對多」關係。因此，樹可以視為圖的一種特例。由此可知，樹的走訪也是圖的走訪的一種特例。

圖和樹都可以透過搜尋演算法來走訪。圖的走訪方式也分為兩種：廣度優先走訪與深度優先走訪。

9.3.1   廣度優先走訪¶

廣度優先走訪從某個節點出發，總是優先走訪距離起點最近的頂點，由近而遠逐層向外擴展。如圖 9-9 所示，從左上角的頂點出發，先走訪該頂點的所有相鄰頂點，再走訪下一個頂點的所有相鄰頂點，依此類推，直到走訪完所有頂點。

距離相同的頂點，走訪順序取決於檢查鄰居的順序，因此廣度優先走訪的序列並非只有一種。

1.   演算法實作¶

BFS 通常以佇列實作，程式碼如下。佇列「先進先出」的特性，正好配合 BFS「由近而遠」的走訪順序。

  1. 將起始頂點 startVet 加入佇列，開始執行迴圈。
  2. 每一輪都從佇列前端取出頂點，記錄走訪結果，再將該頂點的所有相鄰頂點加入佇列尾端。
  3. 重複步驟 2.，直到所有頂點都已走訪為止。

為了避免重複走訪頂點，我們使用雜湊集合 visited 記錄已走訪的節點。

Tip

雜湊集合可以視為只儲存 key、不儲存 value 的雜湊表，能以 \(O(1)\) 的時間複雜度對 key 執行新增、刪除、查詢與修改。由於 key 不會重複，雜湊集合常用來移除重複資料。

graph_bfs.py
def graph_bfs(graph: GraphAdjList, start_vet: Vertex) -> list[Vertex]:
    """廣度優先走訪"""
    # 以鄰接串列表示圖，以便取得指定頂點的所有相鄰頂點
    # 頂點走訪順序
    res = []
    # 用雜湊集合記錄已走訪的頂點
    visited = set[Vertex]([start_vet])
    # 使用佇列實作 BFS
    que = deque[Vertex]([start_vet])
    # 從頂點 vet 出發，反覆執行，直到走訪完所有頂點
    while len(que) > 0:
        vet = que.popleft()  # 取出佇列前端的頂點
        res.append(vet)  # 記錄走訪的頂點
        # 走訪此頂點的所有相鄰頂點
        for adj_vet in graph.adj_list[vet]:
            if adj_vet in visited:
                continue  # 略過已走訪的頂點
            que.append(adj_vet)  # 只將尚未走訪的頂點加入佇列
            visited.add(adj_vet)  # 將此頂點標記為已走訪
    # 回傳頂點的走訪順序
    return res

/* 廣度優先走訪 */
// 以鄰接串列表示圖，以便取得指定頂點的所有相鄰頂點
vector<Vertex *> graphBFS(GraphAdjList &graph, Vertex *startVet) {
    // 頂點走訪順序
    vector<Vertex *> res;
    // 用雜湊集合記錄已走訪的頂點
    unordered_set<Vertex *> visited = {startVet};
    // 使用佇列實作 BFS
    queue<Vertex *> que;
    que.push(startVet);
    // 從頂點 vet 出發，反覆執行，直到走訪完所有頂點
    while (!que.empty()) {
        Vertex *vet = que.front();
        que.pop();          // 取出佇列前端的頂點
        res.push_back(vet); // 記錄走訪的頂點
        // 走訪此頂點的所有相鄰頂點
        for (auto adjVet : graph.adjList[vet]) {
            if (visited.count(adjVet))
                continue;            // 略過已走訪的頂點
            que.push(adjVet);        // 只將尚未走訪的頂點加入佇列
            visited.emplace(adjVet); // 將此頂點標記為已走訪
        }
    }
    // 回傳頂點的走訪順序
    return res;
}

/* 廣度優先走訪 */
// 以鄰接串列表示圖，以便取得指定頂點的所有相鄰頂點
List<Vertex> graphBFS(GraphAdjList graph, Vertex startVet) {
    // 頂點走訪順序
    List<Vertex> res = new ArrayList<>();
    // 用雜湊集合記錄已走訪的頂點
    Set<Vertex> visited = new HashSet<>();
    visited.add(startVet);
    // 使用佇列實作 BFS
    Queue<Vertex> que = new LinkedList<>();
    que.offer(startVet);
    // 從頂點 vet 出發，反覆執行，直到走訪完所有頂點
    while (!que.isEmpty()) {
        Vertex vet = que.poll(); // 取出佇列前端的頂點
        res.add(vet);            // 記錄走訪的頂點
        // 走訪此頂點的所有相鄰頂點
        for (Vertex adjVet : graph.adjList.get(vet)) {
            if (visited.contains(adjVet))
                continue;        // 略過已走訪的頂點

            visited.add(adjVet); // 將此頂點標記為已走訪
        }
    }
    // 回傳頂點的走訪順序
    return res;
}

/* 廣度優先走訪 */
// 以鄰接串列表示圖，以便取得指定頂點的所有相鄰頂點
List<Vertex> GraphBFS(GraphAdjList graph, Vertex startVet) {
    // 頂點走訪順序
    List<Vertex> res = [];
    // 用雜湊集合記錄已走訪的頂點
    HashSet<Vertex> visited = [startVet];
    // 使用佇列實作 BFS
    Queue<Vertex> que = new();
    que.Enqueue(startVet);
    // 從頂點 vet 出發，反覆執行，直到走訪完所有頂點
    while (que.Count > 0) {
        Vertex vet = que.Dequeue(); // 取出佇列前端的頂點
        res.Add(vet);               // 記錄走訪的頂點
        foreach (Vertex adjVet in graph.adjList[vet]) {
            if (visited.Contains(adjVet)) {
                continue;          // 略過已走訪的頂點
            }
            que.Enqueue(adjVet);   // 只將尚未走訪的頂點加入佇列
            visited.Add(adjVet);   // 將此頂點標記為已走訪
        }
    }

    // 回傳頂點的走訪順序
    return res;
}
graph_bfs.go
/* 廣度優先走訪 */
// 以鄰接串列表示圖，以便取得指定頂點的所有相鄰頂點
func graphBFS(g *graphAdjList, startVet Vertex) []Vertex {
    // 頂點走訪順序
    res := make([]Vertex, 0)
    // 用雜湊集合記錄已走訪的頂點
    visited := make(map[Vertex]struct{})
    visited[startVet] = struct{}{}
    // 以切片模擬佇列，用來實作 BFS
    queue := make([]Vertex, 0)
    queue = append(queue, startVet)
    // 從頂點 vet 出發，反覆執行，直到走訪完所有頂點
    for len(queue) > 0 {
        // 取出佇列前端的頂點
        vet := queue[0]
        queue = queue[1:]
        // 記錄走訪的頂點
        res = append(res, vet)
        // 走訪此頂點的所有相鄰頂點
        for _, adjVet := range g.adjList[vet] {
            _, isExist := visited[adjVet]
            // 只將尚未走訪的頂點加入佇列
            if !isExist {
                queue = append(queue, adjVet)
                visited[adjVet] = struct{}{}
            }
        }
    }
    // 回傳頂點的走訪順序
    return res
}

/* 廣度優先走訪 */
// 以鄰接串列表示圖，以便取得指定頂點的所有相鄰頂點
func graphBFS(graph: GraphAdjList, startVet: Vertex) -> [Vertex] {
    // 頂點走訪順序
    var res: [Vertex] = []
    // 用雜湊集合記錄已走訪的頂點

    while !que.isEmpty {
        let vet = que.removeFirst() // 取出佇列前端的頂點
        res.append(vet) // 記錄走訪的頂點

        }
    }
    // 回傳頂點的走訪順序

/* 廣度優先走訪 */
// 以鄰接串列表示圖，以便取得指定頂點的所有相鄰頂點
fn graph_bfs(graph: GraphAdjList, start_vet: Vertex) -> Vec<Vertex> {
    // 頂點走訪順序
    let mut res = vec![];
    // 用雜湊集合記錄已走訪的頂點
    let mut visited = HashSet::new();
    visited.insert(start_vet);
    // 使用佇列實作 BFS
    let mut que = VecDeque::new();
    que.push_back(start_vet);
    // 從頂點 vet 出發，反覆執行，直到走訪完所有頂點
    while let Some(vet) = que.pop_front() {
        res.push(vet); // 記錄走訪的頂點

        // 走訪此頂點的所有相鄰頂點
        if let Some(adj_vets) = graph.adj_list.get(&vet) {
            for &adj_vet in adj_vets {
                if visited.contains(&adj_vet) {
                    continue; // 略過已走訪的頂點
                }
                que.push_back(adj_vet); // 只將尚未走訪的頂點加入佇列
                visited.insert(adj_vet); // 將此頂點標記為已走訪
            }
        }
    }
    // 回傳頂點的走訪順序
    res
}

/* 節點佇列結構 */
typedef struct {
    Vertex *vertices[MAX_SIZE];
    int front, rear, size;
} Queue;

/* 建構函式 */
Queue *newQueue() {
    Queue *q = (Queue *)malloc(sizeof(Queue));
    q->front = q->rear = q->size = 0;
    return q;
}

/* 判斷佇列是否為空 */
int isEmpty(Queue *q) {
    return q->size == 0;
}

/* 加入佇列 */
void enqueue(Queue *q, Vertex *vet) {
    q->vertices[q->rear] = vet;
    q->rear = (q->rear + 1) % MAX_SIZE;
    q->size++;
}

/* 移出佇列 */
Vertex *dequeue(Queue *q) {
    Vertex *vet = q->vertices[q->front];
    q->front = (q->front + 1) % MAX_SIZE;
    q->size--;
    return vet;
}

/* 檢查頂點是否已走訪 */
int isVisited(Vertex **visited, int size, Vertex *vet) {
    // 走訪並尋找節點，耗時 O(n)
    for (int i = 0; i < size; i++) {
        if (visited[i] == vet)
            return 1;
    }
    return 0;
}

/* 廣度優先走訪 */
// 以鄰接串列表示圖，以便取得指定頂點的所有相鄰頂點
void graphBFS(GraphAdjList *graph, Vertex *startVet, Vertex **res, int *resSize, Vertex **visited, int *visitedSize) {
    // 使用佇列實作 BFS
    Queue *queue = newQueue();
    enqueue(queue, startVet);
    visited[(*visitedSize)++] = startVet;
    // 從頂點 vet 出發，反覆執行，直到走訪完所有頂點
    while (!isEmpty(queue)) {
        Vertex *vet = dequeue(queue); // 取出佇列前端的頂點
        res[(*resSize)++] = vet;      // 記錄走訪的頂點
        // 走訪此頂點的所有相鄰頂點
        AdjListNode *node = findNode(graph, vet);
        while (node != NULL) {
            // 略過已走訪的頂點
            if (!isVisited(visited, *visitedSize, node->vertex)) {
                enqueue(queue, node->vertex);             // 只將尚未走訪的頂點加入佇列
                visited[(*visitedSize)++] = node->vertex; // 將此頂點標記為已走訪
            }
            node = node->next;
        }
    }
    // 釋放記憶體
    free(queue);
}

fun graphBFS(graph: GraphAdjList, startVet: Vertex): MutableList<Vertex?> {
    // 頂點走訪順序
    val res = mutableListOf<Vertex?>()
    // 用雜湊集合記錄已走訪的頂點
    val visited = HashSet<Vertex>()
    visited.add(startVet)
    // 使用佇列實作 BFS
    val que = LinkedList<Vertex>()
    que.offer(startVet)
    // 從頂點 vet 出發，反覆執行，直到走訪完所有頂點
    while (!que.isEmpty()) {
        val vet = que.poll() // 取出佇列前端的頂點
        res.add(vet)         // 記錄走訪的頂點
        // 走訪此頂點的所有相鄰頂點
        for (adjVet in graph.adjList[vet]!!) {
            if (visited.contains(adjVet))
                continue        // 略過已走訪的頂點
            que.offer(adjVet)   // 只將尚未走訪的頂點加入佇列
            visited.add(adjVet) // 將此頂點標記為已走訪
        }
    }
    // 回傳頂點的走訪順序
    return res
}

  # 從頂點 vet 出發，反覆執行，直到走訪完所有頂點
  while que.length > 0
    vet = que.shift # 取出佇列前端的頂點

      visited.add(adj_vet) # 將此頂點標記為已走訪
    end
  end

  res

  1. 初始化佇列與集合

    圖中用 res 記錄已離開佇列的頂點，用佇列 que 存放等待處理的頂點（由隊首寫到隊尾），用集合 visited 記錄已發現的頂點。頂點一進入佇列就記入 visited，不必等到離開，因此還在排隊的頂點也不會重複進入佇列。佇列左側的空格是頂點離開後留下的位置，並非空元素。visited 的橫向排列及下文的大括號都不代表集合內部有順序。先將 0 加入佇列，標為已發現：res=[]，que=[0]，visited={0}，此時還沒有頂點離開佇列。

  2. 處理頂點 0

    從隊首取出頂點 0，加到 res 的末端。依序檢查 0 的鄰居 1、3。兩者都還沒被發現，依序加到佇列 que 的隊尾，並記入 visited。目前的結果為 res=[0]，等待處理的佇列為 que=[1, 3]，已發現的頂點集合為 visited={0, 1, 3}。

  3. 處理頂點 1

    從隊首取出頂點 1，加到 res 的末端。在 1 的鄰居 0、2、4 中，0 已在 visited 裡，因此略過。將新發現的鄰居 2、4 依序加入佇列，並記入 visited。目前的結果為 res=[0, 1]，等待處理的佇列為 que=[3, 2, 4]，已發現的頂點集合為 visited={0, 1, 3, 2, 4}。

  4. 處理頂點 3

    從隊首取出頂點 3，加到 res 的末端。在 3 的鄰居 0、4、6 中，0 和 4 都已在 visited 裡，因此略過，只將新發現的鄰居 6 加入佇列，並記入 visited。上一步處理頂點 1 時已發現頂點 4，並將它加入佇列。即使 4 仍在排隊，也不會重複加入。目前的結果為 res=[0, 1, 3]，等待處理的佇列為 que=[2, 4, 6]，已發現的頂點集合為 visited={0, 1, 3, 2, 4, 6}。

  5. 處理頂點 2

    從隊首取出頂點 2，加到 res 的末端。在 2 的鄰居 1、5 中，1 已在 visited 裡，因此略過。將新發現的鄰居 5 加入佇列，並記入 visited。目前的結果為 res=[0, 1, 3, 2]，等待處理的佇列為 que=[4, 6, 5]，已發現的頂點集合為 visited={0, 1, 3, 2, 4, 6, 5}。

  6. 處理頂點 4

  7. 處理頂點 6

    從隊首取出頂點 6，加到 res 的末端。它的鄰居 3、7 都已在 visited 裡（上一步處理 4 時已發現 7，並將它加入佇列），這一輪沒有新鄰居加入佇列。目前的結果為 res=[0, 1, 3, 2, 4, 6]，等待處理的佇列為 que=[5, 7]，已發現的頂點集合為 visited={0, 1, 3, 2, 4, 6, 5, 7}。

  8. 處理頂點 5

    從隊首取出頂點 5，加到 res 的末端。在 5 的鄰居 2、4、8 中，2、4 已在 visited 裡，因此略過。將最後一個新發現的鄰居 8 加入佇列，並記入 visited。至此九個頂點都已發現，佇列裡的 7、8 仍在等待處理。目前的結果為 res=[0, 1, 3, 2, 4, 6, 5]，等待處理的佇列為 que=[7, 8]，已發現的頂點集合為 visited={0, 1, 3, 2, 4, 6, 5, 7, 8}。

距離相同的頂點，走訪順序取決於檢查鄰居的順序，因此廣度優先走訪的序列並非只有一種。

廣度優先走訪的順序只有一種嗎？

不只一種。廣度優先走訪只要求「由近而遠」，距離相同的頂點可以任意調整走訪順序。以圖 9-10 為例，頂點 \(1\)、\(3\) 的走訪順序可以互換，頂點 \(2\)、\(4\)、\(6\) 的走訪順序也可以任意調整。

2.   複雜度分析¶

時間複雜度：每個頂點都會進出佇列各一次，耗時 \(O(|V|)\)。走訪相鄰頂點時，由於是無向圖，每條邊都會走訪 \(2\) 次，耗時 \(O(2|E|)\)。總共耗時 \(O(|V| + |E|)\)。

空間複雜度：串列 res、雜湊集合 visited 與佇列 que 各自最多存放 \(|V|\) 個頂點，所需空間為 \(O(|V|)\)。

9.3.2   深度優先走訪¶

深度優先走訪會沿著一條路一直走到底，無路可走時才回頭。如圖 9-11 所示，從左上角的頂點出發，每次走訪目前頂點的一個相鄰頂點，走到盡頭就返回，再沿另一條路走到底並返回。如此反覆，直到走訪完所有頂點。

/* 深度優先走訪的輔助函式 */
func dfs(g *graphAdjList, visited map[Vertex]struct{}, res *[]Vertex, vet Vertex) {
    // append 會回傳新的切片參照，必須將它重新指派給原變數
    *res = append(*res, vet)
    visited[vet] = struct{}{}
    // 走訪此頂點的所有相鄰頂點
    for _, adjVet := range g.adjList[vet] {
        _, isExist := visited[adjVet]
        // 遞迴走訪相鄰頂點
        if !isExist {
            dfs(g, visited, res, adjVet)
        }

/* 檢查頂點是否已走訪 */
int isVisited(Vertex **res, int size, Vertex *vet) {
    // 走訪並尋找節點，耗時 O(n)
    for (int i = 0; i < size; i++) {
        if (res[i] == vet) {
            return 1;
        }
    }
    return 0;
}

/* 深度優先走訪的輔助函式 */
void dfs(GraphAdjList *graph, Vertex **res, int *resSize, Vertex *vet) {
    // 記錄走訪的頂點
    res[(*resSize)++] = vet;
    // 走訪此頂點的所有相鄰頂點
    AdjListNode *node = findNode(graph, vet);
    while (node != NULL) {
        // 略過已走訪的頂點
        if (!isVisited(res, *resSize, node->vertex)) {
            // 遞迴走訪相鄰頂點
            dfs(graph, res, resSize, node->vertex);
        }
        node = node->next;
    }
}

深度優先走訪的演算法流程如圖 9-12 所示。

  • 直虛線代表向下遞迴，表示展開一次新的遞迴呼叫，走訪新頂點。
  • 曲虛線代表向上回溯，表示這次遞迴呼叫已返回，回到發出呼叫的位置。

圖解 9-12：圖的深度優先走訪：操作過程

從起點出發，沿尚未走訪的鄰居遞迴深入。無路可走時，就回溯到上一層的頂點，繼續探索其餘分支。

這張無向連通圖有頂點 0 至 6，共有 6 條無向邊：0—1、0—3、1—2、2—5、4—5、5—6。從起點 0 開始走訪。本例在頂點 0 先檢查鄰居 1 再檢查 3，在頂點 5 先檢查鄰居 4 再檢查 6。

灰色實線是原圖中的無向邊。藍色直虛線表示遞迴進入尚未走訪的鄰居，橘色弧形虛線表示呼叫結束後返回，兩者都不是新加入圖中的邊。概覽圖中括號內的編號 1 至 14 表示 7 次進入與 7 次返回，包含進入起始呼叫與最外層呼叫返回。這些數字是事件編號，與頂點編號不同。

過程圖以 11 個步驟呈現這 14 個事件。第 1 至 8 步分別對應事件 1 至 8，第 9 步合併事件 9 至 11，第 10 步對應事件 12，第 11 步合併事件 13 至 14。

  1. 走訪起點 0

    過程圖用 res 依首次走訪順序記錄頂點，用 visited 避免重複走訪。大括號只列出集合成員，不表示順序。起始呼叫進入頂點 0（事件 1），先將 0 記入 res 和 visited，再依序檢查鄰居 1、3。目前的遞迴呼叫路徑為 0，res = [0]，visited = {0}。0 的鄰居 1、3 都還沒走訪。

  2. 從 0 深入 1

    頂點 0 有兩個尚未走訪的鄰居 1、3。本例先遞迴進入 1（事件 2），將首次走訪的 1 記入 res 和 visited，等 1 的分支返回後再探索鄰居 3。目前的遞迴呼叫路徑為 0→1，res = [0, 1]，visited = {0, 1}。

  3. 從 1 深入 2

    頂點 1 的鄰居 0 已在 visited 裡，因此略過 0，遞迴進入尚未走訪的鄰居 2（事件 3），將 2 記入 res 和 visited。目前的遞迴呼叫路徑為 0→1→2，res = [0, 1, 2]，visited = {0, 1, 2}。

  4. 從 2 深入 5

    頂點 2 的鄰居 1 已走訪過，接著遞迴進入尚未走訪的鄰居 5（事件 4），將 5 記入 res 和 visited。目前的遞迴呼叫路徑為 0→1→2→5，res = [0, 1, 2, 5]，visited = {0, 1, 2, 5}。5 還有兩個尚未走訪的鄰居 4、6。

  5. 從 5 深入 4

    頂點 5 的鄰居 2 已走訪過，剩下的鄰居 4、6 都還沒走訪。本例先遞迴進入 4（事件 5），將 4 記入 res 和 visited。目前的遞迴呼叫路徑為 0→1→2→5→4，res = [0, 1, 2, 5, 4]，visited = {0, 1, 2, 4, 5}。頂點 3、6 尚未走訪。

  6. 從 4 返回 5

    頂點 4 唯一的鄰居 5 已走訪過，因此結束對 4 的呼叫，返回 5（事件 6）。5 已走訪過，返回時不會再將它寫入 res。目前的遞迴呼叫路徑為 0→1→2→5，res 仍是 [0, 1, 2, 5, 4]，visited 仍是 {0, 1, 2, 4, 5}。5 的鄰居 6 和 0 的鄰居 3 都還沒走訪。

  7. 從 5 深入 6

    回到頂點 5 後，繼續檢查剩下的鄰居 6。6 還沒走訪，因此遞迴進入 6（事件 7），並將 6 記入 res 和 visited。目前的遞迴呼叫路徑為 0→1→2→5→6，res = [0, 1, 2, 5, 4, 6]，visited = {0, 1, 2, 4, 5, 6}。整張圖只剩下頂點 3 還沒走訪。

  8. 從 6 返回 5

    頂點 6 唯一的鄰居 5 已走訪過，因此結束對 6 的呼叫，返回 5（事件 8）。這一步沒有走訪新的頂點。目前的遞迴呼叫路徑為 0→1→2→5，res = [0, 1, 2, 5, 4, 6]，visited = {0, 1, 2, 4, 5, 6}。5 的鄰居 2、4、6 都已處理完畢。

  9. 逐層返回起點 0

    5、2、1 的鄰居都已檢查完畢，沿 5→2→1→0 逐層回溯。事件 9 從 5 返回 2，事件 10 從 2 返回 1，事件 11 從 1 返回 0。目前的遞迴呼叫路徑為 0，res = [0, 1, 2, 5, 4, 6]，visited = {0, 1, 2, 4, 5, 6}。起點 0 還有通往鄰居 3 的分支尚待探索。

  10. 從 0 深入 3

    從 0 通往鄰居 1 的分支已全部返回，接著進入尚未走訪的鄰居 3（事件 12），將 3 記入 res 和 visited。目前的遞迴呼叫路徑為 0→3，res = [0, 1, 2, 5, 4, 6, 3]，visited = {0, 1, 2, 3, 4, 5, 6}。7 個頂點都已走訪，還要等最後的呼叫返回。

  11. 結束所有遞迴呼叫

    頂點 3 唯一的鄰居 0 已走訪過，先從 3 返回 0（事件 13）。此時起點 0 的所有鄰居也已處理完畢，最外層的起始呼叫隨之返回（事件 14），深度優先走訪至此結束。呼叫堆疊已空。最後 res = [0, 1, 2, 5, 4, 6, 3]，visited = {0, 1, 2, 3, 4, 5, 6}。圖中起點 0 上有橘色的返回標記。

res 只記錄第一次走訪的頂點，回溯時不加入頂點。檢查鄰居的順序改變，深度優先走訪的序列也會改變。即使所有頂點都已走訪，仍須完成其餘遞迴呼叫。

深度優先走訪的順序只有一種嗎？

和廣度優先走訪一樣，深度優先走訪的順序也不只一種。從某個頂點出發時，可以先往任一方向探索。無論如何調整相鄰頂點的順序，都仍是深度優先走訪。

以樹的走訪為例，「根 \(\rightarrow\) 左 \(\rightarrow\) 右」、「左 \(\rightarrow\) 根 \(\rightarrow\) 右」、「左 \(\rightarrow\) 右 \(\rightarrow\) 根」分別對應前序、中序、後序走訪。雖然三者的走訪優先順序不同，但都屬於深度優先走訪。

2.   複雜度分析¶

時間複雜度：每個頂點都會走訪 \(1\) 次，耗時 \(O(|V|)\)。每條邊都會走訪 \(2\) 次，耗時 \(O(2|E|)\)。總共耗時 \(O(|V| + |E|)\)。

空間複雜度：串列 res 與雜湊集合 visited 各自最多存放 \(|V|\) 個頂點，遞迴深度最多為 \(|V|\)，因此所需空間為 \(O(|V|)\)。
